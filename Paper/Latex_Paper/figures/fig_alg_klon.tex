\captionof{algorithm}{Rhyme verification of a Klon-Paed poem
(\texttt{check\_klon}).}
\vspace{2pt}
\begin{algorithmic}[1]
\REQUIRE poem $T$ with waks separated by whitespace; type $k\in\{4,8\}$
\ENSURE list of rhyme errors (empty $\Leftrightarrow$ poem is valid)
\STATE $W\gets$ split $T$ into waks, grouped into stanzas of four
\FOR{each stanza $(w_1,w_2,w_3,w_4)$}
  \STATE $S_i\gets$ \textsc{ssg}($w_i$), $i=1,\ldots,4$ \COMMENT{syllables}
  \FOR{each rule $\rho\in\{\mathrm{r1},\mathrm{r2},\mathrm{r3},\mathrm{rX}\}$}
    \STATE $(a,B_\rho)\gets$ source syllable and target set of rule $\rho$
    \IF{no $b\in B_\rho$ satisfies \textsc{is\_sumpus}($a,b$)}
      \STATE record an error for rule $\rho$
    \ENDIF
  \ENDFOR
\ENDFOR
\RETURN the list of errors
\end{algorithmic}
